# Daily backup script
# Run with: Loco.exe run backup.txt

notify "Starting daily backup..."

# Backup documents
backup C:\Users\%USERNAME%\Documents D:\Backup\Documents

# Backup desktop
backup C:\Users\%USERNAME%\Desktop D:\Backup\Desktop

# Backup pictures
backup C:\Users\%USERNAME%\Pictures D:\Backup\Pictures

notify "Backup completed!"
